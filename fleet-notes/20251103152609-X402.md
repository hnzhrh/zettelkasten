```java
package ai.saharalabs.hive.x402.config;  
  
import com.coinbase.x402.client.HttpFacilitatorClient;  
import com.coinbase.x402.server.PaymentFilter;  
import jakarta.servlet.Filter;  
import java.math.BigInteger;  
import java.util.Map;  
import org.springframework.beans.factory.annotation.Value;  
import org.springframework.boot.web.servlet.FilterRegistrationBean;  
import org.springframework.context.annotation.Bean;  
import org.springframework.context.annotation.Configuration;  
import org.springframework.core.Ordered;  
  
@Configuration  
public class X402FilterRegistration {  
  
  @Value("${hive.receiver.address}")  
  private String payToAddress;  
  
  @Value("${hive.facilitator.url}")  
  private String facilitatorUrl;  
  
  // 1. Define paths that require payment and their prices  
  Map<String, BigInteger> priceTable = Map.of(  
      "/apis/weather/v1/huining", BigInteger.valueOf(1000), // 1000 wei  
      "/content/exclusive", BigInteger.valueOf(500)  
  );  
  
  @Bean  
  public FilterRegistrationBean<Filter> paymentFilter() {  
    PaymentFilter filter = new PaymentFilter(payToAddress, priceTable,  
        new HttpFacilitatorClient(facilitatorUrl));  
    FilterRegistrationBean<Filter> registrationBean = new FilterRegistrationBean<>();  
    registrationBean.setFilter(filter);  
    registrationBean.addUrlPatterns("/*");  
    registrationBean.setName("paymentFilter");  
    registrationBean.setOrder(Ordered.LOWEST_PRECEDENCE);  
    return registrationBean;  
  }  
}
```
